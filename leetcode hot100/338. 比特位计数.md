# LeetCode 338. 比特位计数

## 1. 思路：

1. 状态转移：dp[i] = dp[i - pre] + 1
2. pre是i的左边第一个2次幂，比如5，6，7的pre是4
3. 6的二进制'1'个数=（6-4）的二进制'1'的个数+1，110 = 100 + 010

---

## 2. 背诵

```cpp
class Solution {
public:
    vector<int> countBits(int n) {
        vector<int>dp(n + 1, 0);
        if(n == 0){
            return dp;
        }
        for(int i = 1; i <= n; i++){
            if(i % 2 == 0)
                dp[i] = 1;
        }
        dp[1] = 1;
        // 状态转移方程：i这个数的二进制'1'个数等于，i - pre二进制'1'的个数 + 1;
        int pre = 2;
        for(int i = 3; i <= n; i++){
            if(pre * 2 == i){
                pre *= 2;
            }
            dp[i] = dp[i - pre] + 1;
        }
        return dp;
    }
};
