# LeetCode 347. 前 K 个高频元素 399. 除法求值

## 1. 思路：

1. 已知a/b，b/c，求a/c就是条件相乘
2. 构造一个有向图，节点是a，b，c，边权就是a/b的值...

---

## 2. 背诵

```cpp
class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        // 哈希表记录每个数出现次数
        unordered_map<int, int> cnt;
        int max_cnt = 0;
        for(auto num : nums){
            cnt[num]++;
            max_cnt = max(max_cnt, cnt[num]);
        }

        // 构造桶排序[出现次数，数字]
        vector<vector<int>> bucket(max_cnt + 1);
        for(auto x : cnt){
            bucket[x.second].push_back(x.first);
        }
        
        vector<int> ans;
        for(int i = max_cnt; i >= 0 && ans.size() < k; i--){
            for(int j = 0; j < bucket[i].size(); j++){
               ans.push_back(bucket[i][j]);
                if(ans.size() == k){
                    break;
                }
            }          
        }
        return ans;
    }
};
