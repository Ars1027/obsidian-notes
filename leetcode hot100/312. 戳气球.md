# LeetCode 312. 戳气球

## 1. 思路：

1. 设有一个开区间(i, j)，dp[i]  [j] 表示在这个区间内可以获得最多的收益
2. 设k是i到j之间最后一个戳破的气球，dp[i]  [j] = dp[i]   [k] + do[k]   [j] + nums[k - 1] * nums[k] * nums[k + 1]
---

## 2. 背诵

```cpp
lass Solution {
public:
    int maxCoins(vector<int>& nums) {
        // 两端补上1，处理边界情况
        nums.insert(nums.begin(), 1);
        nums.push_back(1);
        
        int N = nums.size();                          
        vector<vector<int>> dp(N, vector<int>(N, 0));

  

        for (int len = 2; len < N; len++) {           // 开区间长度
            for (int i = 0; i + len < N; i++) {       // 左端点
                int j = i + len;                      // 右端点
                for (int k = i + 1; k < j; k++) {     // k 是 (i, j) 内最后一个
                    dp[i][j] = max(dp[i][j], dp[i][k] + dp[k][j] 
			                    + nums[i] * nums[k] * nums[j]);
                }
            }
        }
        return dp[0][N - 1];  
    }
};
