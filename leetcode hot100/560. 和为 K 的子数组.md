# LeetCode 23. 560. 和为 K 的子数组

## 1. 思路：

### 方法 1 前缀和数组+哈希表 （超时）

```cpp
class Solution {

public:
    int subarraySum(vector<int>& nums, int k) {
        int n = nums.size();
        vector<int> sum(n + 1, 0);
        for (int i = 0; i < n; i++) {
            sum[i + 1] = sum[i] + nums[i];
        }

        unordered_map<int, int> m;
        m[k] = 0;

        for (int i = n; i >= 1; i--) {
            for (int j = 1; j <= i; j++) {
                m[sum[i] - sum[j - 1]]++;
            }
        }
        return m[k];
    }
};
```

## 3. 背诵

```cpp
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int, int> mp;

        mp[0] = 1;  // 前缀和为 0 出现过 1 次

        int pre = 0; // 前置和
        int ans = 0;

        for (int x : nums) {
            pre += x;
            
            auto it = mp.find(pre - k); // 找到之前有多少个前缀和等于 pre - k
            if (it != mp.end()) {
                ans += it->second;
            }
            mp[pre]++;
        }
        return ans;

    }

};
