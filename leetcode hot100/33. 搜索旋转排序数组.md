# LeetCode 33. 搜索旋转排序数组

## 1. 思路：

- 翻转之后，数组最后一个就是分界点
- 比他大，则在前半段的大区间
- 比他小，则在后半段的小区间


---

## 2. 背诵

```cpp
class Solution {
public:
	// 判断mid和target的位置关系，如果mid在右边，则true，否则false
    bool check(int mid, vector<int>& nums, int target, int last) {
        if (nums[mid] > last) {
            // mid 在大区间
            if (target > last) {
                // target 也在大区间
                if (nums[mid] >= target) {
                    return true;   // mid 在 target 右侧（含）
                } else {
                    return false;  // mid 在 target 左侧
                }
            } else {
                // target 在小区间，mid 在大区间 ⇒ mid 在 target 左侧
                return false;
            }
        } else {
            // mid 在小区间
            if (target > last) {
                // target 在大区间，mid 在小区间 ⇒ mid 在 target 右侧
                return true;
            } else {
                // target 也在小区间
                if (nums[mid] >= target) {
                    return true;   // mid 在 target 右侧（含）
                } else {
                    return false;  // mid 在 target 左侧
                }
            }
        }
    }
  
    int search(vector<int>& nums, int target) {

        int last = nums.back();
        int l = 0, r = nums.size() - 1;
        while (l <= r) {

            int mid = l + (r - l) / 2;
            if (nums[mid] == target) {
                return mid;
            }
            if (check(mid, nums, target, last)) {
                r = mid - 1;   // mid 在 target 右侧，往左找
            } else {
                l = mid + 1;   // mid 在 target 左侧，往右找
            }
        }
        return -1;
    }
};
