# LeetCode 399. 除法求值

## 1. 思路：

1. 已知a/b，b/c，求a/c就是条件相乘
2. 构造一个有向图，节点是a，b，c，边权就是a/b的值...

---

## 2. 背诵

```cpp
class Solution {
public:
    //全局变量2个
    vector<double> res;
    bool Nofind;
    
    // DFS：从 val 出发，尝试走到 target
    // 参数说明：
    //   g      邻接表（引用传递，避免拷贝整张图）
    //   visit  访问标记，visit[node]==1 表示当前这条路径上已经走过 node
    //   val    当前所在节点
    //   target 本轮 query 的目标节点
    //   path   累积的权值乘积
    void dfs(unordered_map< string , vector<pair<string,double>> >& g, unordered_map<string,int>& visit,string val, const string& target, const double& path){
        //如果节点已经相连接，那没 必要再dfs搜索了，直接返回
        if( Nofind == false )
            return;
        // 终止条件：当前节点就是目标节点
        // 此时 path 就是“起点到终点”路径上所有边权的连乘，即答案
        if( val == target){
            Nofind = false;          // 标记已找到，让其他递归分支尽快返回
            res.push_back(path);     // 把答案收集起来
            return;
        }  
        
        // 遍历 val 的所有邻居
        for(int j = 0;j<g[val].size(); ++j){
            //检查与val相连接的点，是否已经访问过了。没访问过继续dfs
            // g[val][j].first 是邻居节点名
            // visit[...] == 0 表示当前路径上还没走过这个邻居
            if(visit[ g[val][j].first ] == 0 ){
                // 选择：把邻居标记为已访问，表示“沿着这条边走下去”
                visit[ g[val][j].first ] = 1;
                // 递归：走到邻居节点，累积权值乘以当前边的权值
                dfs( g,visit, g[val][j].first, target, path*g[val][j].second );
                // 回溯：撤销选择，把邻居的访问标记清掉
                visit[ g[val][j].first ] = 0;
            }
        }
    }
  
    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values, vector<vector<string>>& queries) {
        //string - string(double)  a连接b(b带上权值)
        unordered_map< string , vector<pair<string,double>> > g;
        // 访问标记：visit[node] == 1 表示当前 DFS 路径上已经走过 node
        unordered_map<string,int> visit;
        //构建无向图，a-b的value是3 的话 ，b-a是3的倒数
        for(int i =0;i<equations.size();++i){
            // a -> b，权值为 values[i]
            g[equations[i][0]].push_back( make_pair(equations[i][1], values[i] ) );
            // b -> a，权值为 1.0 / values[i]
            g[equations[i][1]].push_back( make_pair(equations[i][0], 1.0/values[i] ) );
        }

        //遍历queries，对每一组进行dfs计算结果。
        //如果相连接，就把 路径上的权值相乘就是结果
        for(int i =0; i<queries.size();++i){
            //如果queries[0]是不存在的，直接出结果：-1
            // 起点不在图中，说明它没在任何 equation 里出现过，肯定无解
            if( g.find(queries[i][0]) == g.end() ){
                res.push_back(-1.0);
                continue;   // 跳过本轮 DFS
            }
            //设置一个全局变量，如果进行dfs后，queries[0]到不了queries[1]，Nofind = true;
            // 每轮 query 开始前重置为 true（表示“还没找到”）
            Nofind = true;
            // 起点先标记为已访问，防止自环或走回起点
            visit[queries[i][0]]=1;
            // 从起点出发，初始累积权值为 1
            dfs(g,visit,queries[i][0],queries[i][1],1);
            // 回溯：把起点标记清掉，保证下一轮 query 的环境干净
            visit[queries[i][0]]=0;
            // DFS 跑完 Nofind 仍为 true，说明起点到终点不连通
            // 注意：如果找到了，答案已经在 dfs 内部 push 进 res 了，这里不再重复 push
            if( Nofind )
            res.push_back(-1.0);
        }
        return res;
    }
};
