class Solution {

public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        if(nums.size() == 0 || k == 0){
            return {};
        }
		// 单调队列：从队首到队尾保持严格递减
        // 队首元素就是当前窗口的最大值
        deque<int> d;
        vector<int> ans;
        // j：当前窗口右端点下标，从 0 开始
        // i：当前窗口左端点下标，初始为 1 - k
        //    这样当 j = 0 时 i = 1 - k，窗口 [i, j] 还没完全进入数组
        //    当 j = k - 1 时 i = 0，窗口正好覆盖前 k 个元素
        for(int j = 0, i = 1 - k; j < nums.size(); i++, j++){
            // 窗口右移时，上一轮最左边的元素 nums[i - 1] 滑出窗口
            // 如果它正好是队首（当前窗口的最大值候选），就把它从队首移除
            // 注意必须先把 i > 0 放在前面，避免 i - 1 为负导致下标越界
            if(i > 0 && d.front() == nums[i - 1])
                d.pop_front();
            // 加入新元素 nums[j] 前，弹出队尾所有比它小的元素
            // 因为这些元素不可能是后续窗口的最大值了
            // 这样队列从队首到队尾保持递减
            while(!d.empty() && d.back() < nums[j])
                d.pop_back();
            d.push_back(nums[j]);
            if(i >= 0)
                ans.push_back(d.front());
        }
        return ans;
    }
};
