class Solution {
public:
    vector<vector<int>> res;
    void dfs(int cnt, vector<int> &nums){
        if(cnt == nums.size() - 1){ //中止条件，到数组的最后一位
            res.push_back(nums);
            return;
        }
		// 枚举每一位和后面进行交换的情况
        for(int i = cnt; i < nums.size(); i++){
            swap(nums[i], nums[cnt]);
            dfs(cnt + 1, nums);
            swap(nums[i], nums[cnt]); //回溯
        }
    }

    vector<vector<int>> permute(vector<int>& nums) {
        dfs(0, nums);
        return res;
    }
};
