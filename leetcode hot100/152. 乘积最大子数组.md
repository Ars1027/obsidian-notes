# LeetCode 152. 乘积最大子数组数组的乘积

## 1. 核心思路：

1. 维护当前最大值和当前最小值
2. dp方程如下：
```cpp
	// 最小值可能为负数，但没准下一步乘以一个负数，
	// 当前的最大值就变成最小值，而最小值则变成最大值了。
	maxDP[i + 1] = max(maxDP[i] * A[i + 1], A[i + 1],minDP[i] * A[i + 1]);
	minDP[i + 1] = min(minDP[i] * A[i + 1], A[i + 1],maxDP[i] * A[i + 1]);
	dp[i + 1] = max(dp[i], maxDP[i + 1]);
```
---

## 2. 背诵

```cpp
class Solution {

public:
    int maxProduct(vector<int>& nums) {
        int n = nums.size();
        if(n == 0){
            return 0;
        } else if(n == 1) {
            return nums[0];
        }
        int maxDP = nums[0];
        int minDP = nums[0];
        int ans = nums[0];
        for(int i = 1; i < n; i++){
		    // 保存一下maxDP，待会会被覆盖
            int temp = maxDP;
            // 当前最大 × 当前数，当前数，当前最小×当前数，进行比较
            maxDP = max(max(maxDP * nums[i], nums[i]), minDP * nums[i]);
            // 别忘了维护当前最小值
            // 当前最小 × 当前数，当前数，当前最大×当前数，进行比较
            minDP = min(min(minDP * nums[i], nums[i]), temp * nums[i]);
            ans = max(ans, maxDP);
        }
        return ans;

    }

};
