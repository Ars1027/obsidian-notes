class Trie {

private:

    bool isEnd; //单词是否是结尾了

    Trie *next[26]; //相当于一个26叉树

public:

    // 构造函数

    Trie() {

        isEnd = false;

        memset(next, 0, sizeof(next));

    }

    void insert(string word) {

        Trie *node = this; // 拿一个临时指针 node，从根节点开始往下遍历，避免直接修改 this（根节点）。

        for(char c : word){

            if(node->next[c - 'a'] == NULL)

                node->next[c - 'a'] = new Trie();//使用构造函数初始化节点

            node = node->next[c - 'a'];

        }

        node->isEnd = true; //最后一个字母了

    }

    bool search(string word) {

        Trie *node = this;

        for(char c : word){

            node = node->next[c - 'a'];

            if(node == NULL)

                return false;

        }

        return node->isEnd;

    }

    bool startsWith(string prefix) {

        Trie *node = this;

        for(char c : prefix){

            node = node->next[c - 'a'];

            if(node == NULL)

                return false;

        }

        return true; // 和查找不同，不需要匹配最后一个字母，因为是判断是否有前缀

    }

};
