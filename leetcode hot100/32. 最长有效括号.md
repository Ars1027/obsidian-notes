# LeetCode 32. 最长有效括号

## 思路一：常规括号匹配用栈来做

# 代码

```cpp
class Solution {
public:
    int longestValidParentheses(string s) {
        int ans = 0;

        // 栈中存的是“下标”
        // 作为判断是否是连续成功配对括号的边界
        stack<int> st;
        st.push(-1);

        for (int i = 0; i < (int)s.size(); ++i) {
            if (s[i] == '(') {
                // 左括号：把它的下标压栈，等待后面的右括号来匹配
                st.push(i);
            } else {
                // 右括号：先尝试和栈顶的左括号匹配
                st.pop();
                if (st.empty()) {
                    // 栈空了，说明当前这个右括号无法匹配
                    // 它会把前面的有效括号串“截断”
                    // 所以把当前右括号下标压栈，作为新的起点边界
                    st.push(i);
                } else {
                    // 栈不空，说明当前右括号匹配成功
                    // 当前有效括号串长度 = 当前位置 i - 栈顶下标
                    // 栈顶下标是“当前有效串起点的前一个位置”
                    ans = max(ans, i - st.top());
                }
            }
        }
        return ans;
    }
};
```
