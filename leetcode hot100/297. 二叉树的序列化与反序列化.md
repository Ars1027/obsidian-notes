class Codec {
public:

    string serialize(TreeNode* root) {
        if(root == nullptr)
            return "[]";
        string str = "";
        queue<TreeNode*> q;
        q.push(root);
        
        while(!q.empty()){
            TreeNode* node = q.front();
            q.pop();
            if(node != nullptr){
                str += to_string(node->val)+ ',';
                q.push(node->left);
                q.push(node->right);
            }
            else{
                str += "null,";
            }
        }

        str.pop_back();  // 多余的逗号
        return str;

    }

    TreeNode* deserialize(string data) {
        if (data == "[]") return nullptr;
        // 1. 按逗号切分
        vector<string> tokens;
        stringstream ss(data); // 这段字符串当作一个输入流来读
        string token;
        while (getline(ss, token, ',')) {
            tokens.push_back(token);
        }

        // 2. 建根节点入队
        TreeNode* root = new TreeNode(stoi(tokens[0]));
        queue<TreeNode*> q;
        q.push(root);

        // 3. 队列法反序列化
        int i = 1;
        while (!q.empty() && i < tokens.size()) {
            TreeNode* node = q.front();
            q.pop();


            // 处理左孩子
            if ( tokens[i] != "null") {
                node->left = new TreeNode(stoi(tokens[i]));
                q.push(node->left); // 非空孩子入队，等以后给它挂孩子
            }
            ++i;
            // 处理右孩子
            if ( tokens[i] != "null") {
                node->right = new TreeNode(stoi(tokens[i]));
                q.push(node->right);
            }
            ++i;
        }
        return root;
    }
};
